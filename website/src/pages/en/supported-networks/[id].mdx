export const getStaticProps = async ({ params }) => {
  const networks = await getSupportedNetworks()
  const network = networks.find((n) => n.id === params.id)

if (!network) { return { notFound: true, } }

return { props: { network: { id: network.id, fullName: network.fullName, networkType: network.networkType || 'Unknown', protocol: 'Unknown', chainId: 0, nativeCurrency: 'Unknown', docs: network.docsUrl || '#', }, }, } }

export const getStaticPaths = async () => {
  const networks = await getSupportedNetworks()

const paths = networks.map((network) => ({ params: { id: network.id }, }))

return { paths, fallback: false, } }

export function DynamicTitle({ title }) {
  useEffect(() => {
    document.title = title;
  }, [title]);

return ( <Head> <title>{title}</title> </Head> ); }

export function Time({ variant, minutes, className, ...props }) {
  const { t } = useI18n()
  return (
    <div className={classNames(['flex items-center gap-1 leading-none', className])} {...props}>
      <Clock
        alt={variant === 'reading' ? t('index.time.reading') : t('index.time.duration')}
        variant="fill"
        size={3.5}
      />
      {minutes} {t('index.time.minutes')}
    </div>
  )
}
